cmake_minimum_required(VERSION 3.20)
project(hedera_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hedera_core sim/hedera-core.cc sim/hedera-monitoring.cc sim/hedera-controller.cc sim/hedera-oracle.cc)
target_include_directories(hedera_core PUBLIC sim)
target_compile_options(hedera_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(core-tests tests/core-tests.cc)
target_link_libraries(core-tests PRIVATE hedera_core)
enable_testing()
add_test(NAME core-tests COMMAND core-tests)
add_executable(scheduler-benchmark tests/scheduler-benchmark.cc)
target_link_libraries(scheduler-benchmark PRIVATE hedera_core)
add_executable(controller-tests tests/controller-tests.cc)
target_link_libraries(controller-tests PRIVATE hedera_core)
add_test(NAME controller-tests COMMAND controller-tests)
add_executable(oracle-study tests/oracle-study.cc)
target_link_libraries(oracle-study PRIVATE hedera_core)
add_executable(paper-flow-study tests/paper-flow-study.cc)
target_link_libraries(paper-flow-study PRIVATE hedera_core)
add_executable(paper-core-study tests/paper-core-study.cc)
target_link_libraries(paper-core-study PRIVATE hedera_core)
